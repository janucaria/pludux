# Embed the checked-in Lucide font as a C byte array so it can be loaded with
# ImGui::AddFontFromMemoryTTF (no filesystem access required at runtime).
set(PLUDUX_ICON_FONT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(PLUDUX_ICON_FONT_TTF "${CMAKE_SOURCE_DIR}/data/fonts/lucide.ttf")
set(PLUDUX_ICON_FONT_DATA_HEADER "${PLUDUX_ICON_FONT_DIR}/pludux_icon_font_data.h")
if(
  NOT EXISTS "${PLUDUX_ICON_FONT_DATA_HEADER}"
  OR "${PLUDUX_ICON_FONT_TTF}" IS_NEWER_THAN "${PLUDUX_ICON_FONT_DATA_HEADER}"
)
  file(READ "${PLUDUX_ICON_FONT_TTF}" PLUDUX_ICON_FONT_HEX HEX)
  string(LENGTH "${PLUDUX_ICON_FONT_HEX}" PLUDUX_ICON_FONT_HEX_LENGTH)
  math(EXPR PLUDUX_ICON_FONT_DATA_SIZE "${PLUDUX_ICON_FONT_HEX_LENGTH} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PLUDUX_ICON_FONT_BYTES "${PLUDUX_ICON_FONT_HEX}")
  file(WRITE "${PLUDUX_ICON_FONT_DATA_HEADER}"
    "// Auto-generated by CMake from lucide.ttf. Do not edit.\n"
    "#pragma once\n"
    "inline constexpr unsigned char pludux_icon_font_data[] = {\n${PLUDUX_ICON_FONT_BYTES}\n};\n"
    "inline constexpr unsigned int pludux_icon_font_data_size = ${PLUDUX_ICON_FONT_DATA_SIZE}u;\n"
  )
  unset(PLUDUX_ICON_FONT_HEX)
  unset(PLUDUX_ICON_FONT_BYTES)
endif()

  moving_avg_2_line_cross.json

  bollinger_bands_S.json
  bollinger_bands_LS.json
  rsi.json

  stochastic_rsi.json
)

foreach(PLUDUX_BUILT_IN_STRATEGY_JSON_FILE_NAME
        IN LISTS PLUDUX_BUILT_IN_STRATEGY_JSON_FILE_NAMES)
  set(PLUDUX_BUILT_IN_STRATEGY_JSON_FILE
    "${CMAKE_SOURCE_DIR}/data/backtest/samples/${PLUDUX_BUILT_IN_STRATEGY_JSON_FILE_NAME}"
  )
  if(NOT EXISTS "${PLUDUX_BUILT_IN_STRATEGY_JSON_FILE}")
    message(FATAL_ERROR
      "Missing built-in strategy: ${PLUDUX_BUILT_IN_STRATEGY_JSON_FILE}"
    )
  endif()

  get_filename_component(PLUDUX_BUILT_IN_STRATEGY_NAME
    "${PLUDUX_BUILT_IN_STRATEGY_JSON_FILE_NAME}" NAME_WE
  )
  string(MAKE_C_IDENTIFIER "${PLUDUX_BUILT_IN_STRATEGY_NAME}"
    PLUDUX_BUILT_IN_STRATEGY_NAME_IDENTIFIER
  )

  file(READ "${PLUDUX_BUILT_IN_STRATEGY_JSON_FILE}"
    PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT
  )
  string(REPLACE "\\" "\\\\" PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT
    "${PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT}"
  )
  string(REPLACE "\"" "\\\"" PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT
    "${PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT}"
  )
  string(REPLACE "\n" "\\n" PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT
    "${PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT}"
  )
  string(REPLACE "\r" "\\r" PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT
    "${PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT}"
  )
  string(REPLACE "#" "\\u0023" PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT
    "${PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT}"
  )

  # TODO: this is a bit hacky, replace with C++ #embed when it's widely supported
  target_compile_definitions(${PROJECT_NAME} PRIVATE
    "PLUDUX_BUILT_IN_STRATEGY_${PLUDUX_BUILT_IN_STRATEGY_NAME_IDENTIFIER}=\"${PLUDUX_BUILT_IN_STRATEGY_JSON_CONTENT}\""
  )
endforeach()
